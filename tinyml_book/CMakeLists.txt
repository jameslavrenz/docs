cmake_minimum_required(VERSION 3.18)
project(tinyml_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tinyml_ref STATIC examples/int8_reference/int8_reference.cpp)
target_include_directories(tinyml_ref PUBLIC examples/int8_reference)
target_compile_options(tinyml_ref PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(tinyml_demo examples/int8_reference/demo.cpp)
target_link_libraries(tinyml_demo PRIVATE tinyml_ref)
enable_testing()
add_executable(tinyml_tests tests/test_int8_reference.cpp)
target_link_libraries(tinyml_tests PRIVATE tinyml_ref)
target_compile_options(tinyml_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME integer_reference COMMAND tinyml_tests)

add_library(tinyml_full_classifier STATIC examples/full_classifier/full_classifier.cpp)
target_include_directories(tinyml_full_classifier PUBLIC examples/full_classifier)
target_link_libraries(tinyml_full_classifier PUBLIC tinyml_ref)
add_executable(tinyml_classifier_demo examples/full_classifier/demo.cpp)
target_link_libraries(tinyml_classifier_demo PRIVATE tinyml_full_classifier)
add_executable(tinyml_classifier_tests tests/test_full_classifier.cpp)
target_link_libraries(tinyml_classifier_tests PRIVATE tinyml_full_classifier)
add_test(NAME full_classifier COMMAND tinyml_classifier_tests)
